-- 4️⃣ Animaux (200)
DECLARE @i INT = 1;
WHILE @i <= 200
BEGIN
    INSERT INTO Animal (Name, Sex, SpeciesId, OwnerId, IsAvailable, BirthDate, RIPDate)
    VALUES (
        CONCAT('Animal_', @i),
        CASE WHEN @i % 2 = 0 THEN 0 ELSE 1 END,  -- 0 = Male, 1 = Female
        ((@i - 1) % 20) + 1,                     -- rotation sur 20 espèces
        ((@i - 1) % 5) + 1,                      -- rotation sur 5 propriétaires
        CASE WHEN @i % 7 = 0 THEN 0 ELSE 1 END,  -- certains indisponibles
        DATEADD(DAY, -(@i * 20), GETDATE()),     -- dates de naissance variées
        NULL
    );
    SET @i += 1;
END;
